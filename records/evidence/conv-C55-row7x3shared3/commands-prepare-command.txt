import subprocess,json

(b/'commands-prepare-result.json').write_text(json.dumps({'argv':command,'exit_code':r.returncode,'operator_executed':False},indent=2)+'\n')
print(r.stdout,end=''); print(r.stderr,end='')
raise SystemExit(r.returncode)
